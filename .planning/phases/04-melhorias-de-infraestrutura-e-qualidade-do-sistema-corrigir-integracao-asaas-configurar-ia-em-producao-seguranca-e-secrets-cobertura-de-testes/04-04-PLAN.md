```typescript
import { describe, it, expect } from "vitest";
import { getAsaasConfig } from "@/lib/asaas";

describe("getAsaasConfig", () => {
  it("retorna null quando apiKey ausente", () => {
    const result = getAsaasConfig({ asaasApiKey: null, asaasEnv: null });
    // Without env var ASAAS_API_KEY, should return null
    const envKey = process.env.ASAAS_API_KEY;
    if (!envKey) {
      expect(result).toBeNull();
    }
  });

  it("retorna config com sandbox como default", () => {
    const result = getAsaasConfig({ asaasApiKey: "test-key-123", asaasEnv: null });
    expect(result).not.toBeNull();
    expect(result!.apiKey).toBe("test-key-123");
    expect(result!.baseUrl).toBe("https://api-sandbox.asaas.com");
  });

  it("retorna config com production quando env=production", () => {
    const result = getAsaasConfig({ asaasApiKey: "test-key-123", asaasEnv: "production" });
    expect(result).not.toBeNull();
    expect(result!.baseUrl).toBe("https://api.asaas.com");
  });

  it("ignora apiKey com apenas espacos", () => {
    const result = getAsaasConfig({ asaasApiKey: "   ", asaasEnv: null });
    const envKey = process.env.ASAAS_API_KEY;
    if (!envKey) {
      expect(result).toBeNull();
    }
  });
});
```

```typescript
import { describe, it, expect } from "vitest";
import { signReciboId, verifyReciboSig } from "@/lib/recibo-url";

describe("recibo-url", () => {
  it("sig gerada e valida", () => {
    const sig = signReciboId(42);
    expect(verifyReciboSig("42", sig)).toBe(true);
  });

  it("sig invalida rejeitada", () => {
    expect(verifyReciboSig("42", "wrong-sig")).toBe(false);
  });

  it("sig de IDs diferentes nao sao iguais", () => {
    const sig1 = signReciboId(1);
    const sig2 = signReciboId(2);
    expect(sig1).not.toBe(sig2);
  });

  it("retorna false para id ou sig vazios", () => {
    expect(verifyReciboSig("", "some-sig")).toBe(false);
    expect(verifyReciboSig("42", "")).toBe(false);
  });

  it("sig e deterministica (mesmo id = mesma sig)", () => {
    const sig1 = signReciboId(99);
    const sig2 = signReciboId(99);
    expect(sig1).toBe(sig2);
  });
});
```

```typescript
import { describe, it, expect, vi } from "vitest";

// Mock getSession to avoid cookie dependency
vi.mock("@/lib/session", () => ({
  getSession: vi.fn(),
}));

import { requireAdmin, requireAdminOrThrow } from "@/lib/guards";
import { getSession } from "@/lib/session";

const mockGetSession = vi.mocked(getSession);

describe("requireAdmin", () => {
  it("retorna true para admin", async () => {
    mockGetSession.mockResolvedValue({ id: 1, username: "admin", role: "admin" });
    expect(await requireAdmin()).toBe(true);
  });

  it("retorna false para vendedor", async () => {
    mockGetSession.mockResolvedValue({ id: 2, username: "user", role: "vendedor" });
    expect(await requireAdmin()).toBe(false);
  });

  it("retorna false sem sessao", async () => {
    mockGetSession.mockResolvedValue(null);
    expect(await requireAdmin()).toBe(false);
  });
});

describe("requireAdminOrThrow", () => {
  it("retorna session para admin", async () => {
    mockGetSession.mockResolvedValue({ id: 1, username: "admin", role: "admin" });
    const session = await requireAdminOrThrow();
    expect(session.id).toBe(1);
    expect(session.role).toBe("admin");
  });

  it("lanca erro para vendedor", async () => {
    mockGetSession.mockResolvedValue({ id: 2, username: "user", role: "vendedor" });
    await expect(requireAdminOrThrow()).rejects.toThrow("Acesso negado");
  });

  it("lanca erro sem sessao", async () => {
    mockGetSession.mockResolvedValue(null);
    await expect(requireAdminOrThrow()).rejects.toThrow("Nao autenticado");
  });
});
```

**src/__tests__/env-validation.test.ts:**
```typescript
import { describe, it, expect } from "vitest";
import { z } from "zod";

// We test the schema logic directly (not the env module, which runs at import)
describe("env validation schema", () => {
  const envSchema = z.object({
    DATABASE_URL: z.string().min(1, "DATABASE_URL e obrigatoria"),
    SESSION_SECRET: z.string().default("dev-secret"),
    NODE_ENV: z.enum(["development", "test", "production"]).default("development"),
  });

  it("valida env minima corretamente", () => {
    const result = envSchema.safeParse({ DATABASE_URL: "postgresql://localhost/test" });
    expect(result.success).toBe(true);
  });

  it("rejeita DATABASE_URL vazia", () => {
    const result = envSchema.safeParse({ DATABASE_URL: "" });
    expect(result.success).toBe(false);
  });

  it("rejeita DATABASE_URL ausente", () => {
    const result = envSchema.safeParse({});
    expect(result.success).toBe(false);
  });

  it("aceita NODE_ENV valido", () => {
    const result = envSchema.safeParse({ DATABASE_URL: "pg://x", NODE_ENV: "production" });
    expect(result.success).toBe(true);
    if (result.success) expect(result.data.NODE_ENV).toBe("production");
  });

  it("rejeita NODE_ENV invalido", () => {
    const result = envSchema.safeParse({ DATABASE_URL: "pg://x", NODE_ENV: "staging" });
    expect(result.success).toBe(false);
  });

  it("default SESSION_SECRET para dev-secret quando ausente", () => {
    const result = envSchema.safeParse({ DATABASE_URL: "pg://x" });
    expect(result.success).toBe(true);
    if (result.success) expect(result.data.SESSION_SECRET).toBe("dev-secret");
  });
});
```

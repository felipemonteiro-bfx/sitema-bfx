Create a new file `src/env.ts` that validates environment variables at import time using zod. This file should NOT be imported by middleware.ts (Edge Runtime does not support all Node APIs). It should be imported by server-side code that needs validated env vars.

```typescript
import { z } from "zod";

const isProduction = process.env.NODE_ENV === "production";
const skipValidation = process.env.SKIP_ENV_VALIDATION === "true";

const envSchema = z.object({
  // Obrigatorias
  DATABASE_URL: z.string().min(1, "DATABASE_URL e obrigatoria"),
  SESSION_SECRET: isProduction
    ? z.string().min(32, "SESSION_SECRET precisa ter minimo 32 chars em producao")
    : z.string().default("dev-secret"),
  NODE_ENV: z.enum(["development", "test", "production"]).default("development"),

  // Opcionais — App
  NEXT_PUBLIC_APP_URL: z.string().optional(),
  CRON_SECRET: z.string().optional(),

  // Opcionais — MinIO
  MINIO_ENDPOINT: z.string().optional(),
  MINIO_ACCESS_KEY: z.string().optional(),
  MINIO_SECRET_KEY: z.string().optional(),
  MINIO_BUCKET: z.string().optional(),
  MINIO_PUBLIC_URL: z.string().optional(),

  // Opcionais — Asaas
  ASAAS_API_KEY: z.string().optional(),
  ASAAS_ENV: z.enum(["sandbox", "production"]).default("sandbox"),

  // Opcionais — WhatsApp (Evolution API)
  EVOLUTION_API_URL: z.string().optional(),
  EVOLUTION_API_KEY: z.string().optional(),
  EVOLUTION_INSTANCE: z.string().optional(),

  // Opcionais — AI providers
  GOOGLE_GENERATIVE_AI_API_KEY: z.string().optional(),
  OPENAI_API_KEY: z.string().optional(),
  LITELLM_URL: z.string().optional(),
  LITELLM_KEY: z.string().optional(),
  LITELLM_MODEL: z.string().optional(),
  LITELLM_ENABLE_THINKING: z.string().optional(),

  // Opcionais — Uber Direct
  UBER_DIRECT_CUSTOMER_ID: z.string().optional(),
  UBER_DIRECT_TOKEN: z.string().optional(),
  UBER_PICKUP_ADDRESS_JSON: z.string().optional(),
});

export type Env = z.infer<typeof envSchema>;

function validateEnv(): Env {
  if (skipValidation) {
    return process.env as unknown as Env;
  }
  const parsed = envSchema.safeParse(process.env);
  if (!parsed.success) {
    const errors = parsed.error.flatten().fieldErrors;
    const formatted = Object.entries(errors)
      .map(([key, msgs]) => `  ${key}: ${(msgs || []).join(", ")}`)
      .join("\n");
    throw new Error(`Env vars invalidas:\n${formatted}`);
  }
  return parsed.data;
}

export const env = validateEnv();
```

Key design decisions:
- `SKIP_ENV_VALIDATION` allows CI builds to pass without all env vars (used in Dockerfile prebuild)
- SESSION_SECRET requires 32+ chars only in production; defaults to "dev-secret" in development
- All integration vars (Asaas, Evolution, AI, MinIO, Uber) are optional since features degrade gracefully
- The validation runs at import time — any server file importing `env` gets validated vars
